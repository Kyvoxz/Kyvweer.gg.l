--[[
    Kyvweer.gg.l - Loader Limpio
    Sube este archivo a GitHub y usa:
    loadstring(game:HttpGet("https://raw.githubusercontent.com/Kyvoxz/Kyvweer.gg.l/refs/heads/main/Kyvweer.gg.l.txt"))()
]]

local url = "https://raw.githubusercontent.com/Kyvoxz/Kyvweer.gg.l/refs/heads/main/Kyvweer.gg.l.txt"

local function safeLoad()
    local success, result = pcall(function()
        return game:HttpGet(url, true)
    end)

    if not success or type(result) ~= "string" or #result < 1000 then
        warn("[Kyvweer] Error al descargar el script:", result)
        return false
    end

    local fn, err = loadstring(result)
    if not fn then
        warn("[Kyvweer] Error de sintaxis / loadstring falló:", err)
        return false
    end

    local ok, execErr = pcall(fn)
    if not ok then
        warn("[Kyvweer] Error al ejecutar:", execErr)
        return false
    end

    print("[Kyvweer] Cargado correctamente.")
    return true
end

safeLoad()
